```python
import base64
import struct
from offsetx_apollo_builder.imagery.gates import run_gates


def _image_uri(kind: str, payload: bytes) -> str:
    return f"data:image/{kind};base64," + base64.b64encode(payload).decode()


def test_A21_header_only_png_is_not_accepted_as_an_image():
    fake = (b"\\x89PNG\\r\\n\\x1a\\n" + b"\\x00\\x00\\x00\\rIHDR"
            + struct.pack(">II", 32, 32) + b"\\x00" * 2048)
    assert not run_gates(_image_uri("png", fake)).passed


def test_A21_truncated_gif_never_crashes_image_batch():
    report = run_gates(_image_uri("gif", b"GIF89a"), min_bytes=1)
    assert report.passed is False
    assert report.failures
```
